\begin{framedtext}
○ 제1종일반주거지역 : ( ㄱ )퍼센트 이상 ( ㄴ )퍼센트 이하 \par
○ 제2종일반주거지역 : ( ㄱ )퍼센트 이상 ( ㄷ )퍼센트 이하 \par
○ 제3종일반주거지역 : ( ㄱ )퍼센트 이상 ( ㄹ )퍼센트 이하 \par
\end{framedtext}
\begin{choices}
  \item ㄱ: 50, ㄴ: 100, ㄷ: 150, ㄹ: 200
  \item ㄱ: 50, ㄴ: 200, ㄷ: 250, ㄹ: 300
  \item ㄱ: 100, ㄴ: 200, ㄷ: 250, ㄹ: 300
  \item ㄱ: 100, ㄴ: 250, ㄷ: 300, ㄹ: 350
  \item ㄱ: 200, ㄴ: 250, ㄷ: 300, ㄹ: 350
\end{choices}
