\begin{choices}
  \item ｢도시 및 주거환경정비법｣에 따른 정비사업
  \item ｢지역 개발 및 지원에 관한 법률｣에 따른 지역개발사업
  \item ｢도로법｣에 따른 고속국도 및 일반국도 건설사업
  \item ｢국토의 계획 및 이용에 관한 법률｣에 따른 도시ㆍ군계획시설사업
  \item ｢관광진흥법｣에 따른 관광단지 개발사업
\end{choices}
\end{question}
