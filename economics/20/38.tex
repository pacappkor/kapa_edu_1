{\small\centering
\begin{tabular}{cc|c|c|}
\cline{3-4}
 & & \multicolumn{2}{c|}{B기업} \\
\cline{3-4}
 & & 반납 & 보유 \\
\hline
\multicolumn{1}{|c|}{\multirow{2}{*}{A기업}} & 반납 & (-100, -700) & (600, 0) \\
\cline{2-4}
\multicolumn{1}{|c|}{} & 보유 & (0, 150) & (0, 0) \\
\hline
\end{tabular}\par}

\begin{choices}
  \item (600, 0), (0, 150)
  \item (-100, -700), (0, 0)
  \item (600, 0)
  \item (0, 0)
  \item (-100, -700)
\end{choices}
