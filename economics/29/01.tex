  \item $P^{*}=190,\ Q^{*}=840$
  \item $P^{*}=195,\ Q^{*}=820$
  \item $P^{*}=200,\ Q^{*}=800$
  \item $P^{*}=205,\ Q^{*}=780$
  \item $P^{*}=210,\ Q^{*}=760$
\end{choices}
