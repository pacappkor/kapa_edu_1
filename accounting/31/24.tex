감가상각누계액 & 340,000 & 100,000 \\ \hline
공정가치 & 450,000 & 480,000 \\ \hline
\end{tabular}\par}

  \item ￦10,000
  \item ￦20,000
  \item ￦30,000
  \item ￦40,000
  \item ￦50,000
\end{choices}
