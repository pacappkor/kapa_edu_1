\par{\small\centering
\begin{tabular}{|c|c|c|c|}\hline
구분 & 액면금액 & 발행주식수 & 비고 \\ \hline

우선주자본금 & 500 & 50 & 누적적, 완전참가적, 배당률 ? \\ \hline
\end{tabular}

\begin{choices}
  \item 5 \%
  \item 6 \%
  \item 7 \%
  \item 8 \%
  \item 9 \%
\end{choices}
\end{question}
