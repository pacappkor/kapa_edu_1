{\small\centering
\begin{tabular}{|c|c|c|c|c|}
\hline

\hline
\makecell{보통주자본금} & \makecell{￦5,000} & \makecell{12,000주} & \makecell{￦60,000,000} & \makecell{배당률 3\%} \\
\hline
\makecell{우선주자본금} & \makecell{￦10,000} & \makecell{3,000주} & \makecell{￦30,000,000} & \makecell{배당률 5\%,\newline{}누적적, 완전참가적} \\
\hline
\end{tabular}\par}
